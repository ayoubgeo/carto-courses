Landsat LST Preprocessing and Spectral Index Extraction
10 min

Paste the following code into your GEE code editor and click Run. Read the beginner notes below to understand what each line does.

Landsat LST Preprocessing and Spectral Index Extraction
var cityROI = ee.Geometry.Polygon([
  [[-112.15, 33.35], [-111.85, 33.35], [-111.85, 33.65], [-112.15, 33.65]]
]);

function maskLandsatSR(image) {
  var qa = image.select('QA_PIXEL');
  var cloudBitMask = 1 << 3;
  var shadowBitMask = 1 << 4;
  var mask = qa.bitwiseAnd(cloudBitMask).eq(0)
             .and(qa.bitwiseAnd(shadowBitMask).eq(0));
  return image.updateMask(mask);
}

var landsatCollection = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(cityROI)
  .filterDate('2023-06-01', '2023-08-31')
  .filter(ee.Filter.lt('CLOUD_COVER', 15))
  .map(maskLandsatSR);

var medianImage = landsatCollection.median().clip(cityROI);

var lstCelsius = medianImage.select('ST_B10')
  .multiply(0.00341802).add(149.0).subtract(273.15)
  .rename('LST');

var ndvi = medianImage.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDVI');
var ndbi = medianImage.normalizedDifference(['SR_B6', 'SR_B5']).rename('NDBI');

var processedImage = ee.Image.cat([lstCelsius, ndvi, ndbi]);

Map.centerObject(cityROI, 11);
Map.addLayer(processedImage.select('NDVI'), {min: -0.1, max: 0.7, palette: ['blue', 'white', 'green']}, 'NDVI');
Map.addLayer(processedImage.select('NDBI'), {min: -0.3, max: 0.3, palette: ['white', 'orange', 'red']}, 'NDBI');
Map.addLayer(processedImage.select('LST'), {min: 25, max: 50, palette: ['blue', 'yellow', 'orange', 'red']}, 'Land Surface Temperature (C)');
🔑
Beginner note: This script filters Landsat 8 Collection 2 Level-2 scene composites over a urban study region during summer. It applies quality assessment pixel masking to remove cloud contamination. Surface temperature is converted from raw Kelvin digital numbers to degrees Celsius using official USGS scale factors. Spectral indices NDVI (vegetation) and NDBI (built-up imperviousness) are computed to serve as environmental drivers of urban heat accumulation.

Population Exposure Integration and GeoAI Vulnerability Modeling
var popImage = ee.ImageCollection('WorldPop/GP/100m/pop')
  .filterBounds(cityROI)
  .filter(ee.Filter.date('2020-01-01', '2020-12-31'))
  .mean()
  .clip(cityROI)
  .rename('Population');

function minMaxNormalize(img, region, scale) {
  var stats = img.reduceRegion({
    reducer: ee.Reducer.minMax(),
    geometry: region,
    scale: scale,
    maxPixels: 1e9
  });
  var minVal = ee.Number(stats.get(img.bandNames().get(0).aside(function(b){ return ee.String(b).cat('_min'); })));
  var maxVal = ee.Number(stats.get(img.bandNames().get(0).aside(function(b){ return ee.String(b).cat('_max'); })));
  return img.subtract(minVal).divide(maxVal.subtract(minVal)).clamp(0, 1);
}

var lstNorm = minMaxNormalize(processedImage.select('LST'), cityROI, 30).rename('LST_norm');
var ndbiNorm = minMaxNormalize(processedImage.select('NDBI'), cityROI, 30).rename('NDBI_norm');
var ndviNorm = minMaxNormalize(processedImage.select('NDVI'), cityROI, 30).rename('NDVI_norm');
var popNorm = minMaxNormalize(popImage.select('Population'), cityROI, 100).rename('Pop_norm');

var compositeVulnerability = lstNorm.multiply(0.35)
  .add(ndbiNorm.multiply(0.25))
  .add(popNorm.multiply(0.25))
  .add(ee.Image(1).subtract(ndviNorm).multiply(0.15))
  .rename('Heat_Vulnerability_Index');

var trainingPoints = compositeVulnerability.sample({
  region: cityROI,
  scale: 30,
  numPixels: 1000,
  seed: 42,
  geometries: true
});

var predictors = ee.Image.cat([processedImage.select(['NDVI', 'NDBI']), popImage]);

var classifier = ee.Classifier.smileRandomForest(50).train({
  features: trainingPoints,
  classProperty: 'Heat_Vulnerability_Index',
  inputProperties: ['NDVI', 'NDBI', 'Population']
});

Map.addLayer(popImage, {min: 0, max: 200, palette: ['white', 'purple']}, 'Population Density');
Map.addLayer(compositeVulnerability, {min: 0.2, max: 0.8, palette: ['blue', 'yellow', 'orange', 'darkred']}, 'Heat Vulnerability Index (HVI)');
🔑
Beginner note: This script integrates WorldPop population density data and normalizes environmental and demographic predictors to a uniform 0 to 1 scale. It calculates an additive Heat Vulnerability Index (HVI) combining surface temperature, impervious structures, population density, and vegetation deficit. Training samples are generated from the composite index to demonstrate machine learning regression workflows using Random Forest algorithms in Earth Engine.
