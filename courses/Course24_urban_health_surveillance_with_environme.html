Multi-Sensor Environmental Exposure Mapping
10 min

Paste the following code into your GEE code editor and click Run. Read the beginner notes below to understand what each line does.

Multi-Sensor Environmental Exposure Mapping
// Urban Health Surveillance: Multi-Sensor Environmental Exposure Mapping
// Target Region: London Metropolitan Area
var urbanPoint = ee.Geometry.Point([-0.1278, 51.5074]);
var roi = urbanPoint.buffer(25000).bounds();

// Time window for summer season
var startDate = '2023-06-01';
var endDate = '2023-08-31';

// 1. Sentinel-5P Nitrogen Dioxide (NO2) Exposure
var s5p = ee.ImageCollection('COPERNICUS/S5P/OFFL/L3_NO2')
  .filterBounds(roi)
  .filterDate(startDate, endDate)
  .select('tropospheric_NO2_column_number_density');
var meanNO2 = s5p.mean().clip(roi);

// 2. MODIS Land Surface Temperature (LST) Exposure
var modis = ee.ImageCollection('MODIS/061/MOD11A2')
  .filterBounds(roi)
  .filterDate(startDate, endDate)
  .select('LST_Day_1km');

var convertToCelsius = function(img) {
  var celsius = img.multiply(0.02).subtract(273.15);
  return celsius.copyProperties(img, ['system:time_start']);
};
var meanLST = modis.map(convertToCelsius).mean().clip(roi);

// 3. GHSL Built-up Intensity
var ghsl = ee.Image('JRC/GHSL/P2023A/GHS_BUILT_S/2020')
  .select('built_surface')
  .clip(roi);

// Center Map
Map.centerObject(roi, 10);

// Visualization Parameters
var vizNO2 = {min: 0.00003, max: 0.00015, palette: ['blue', 'yellow', 'red']};
var vizLST = {min: 18, max: 35, palette: ['blue', 'green', 'yellow', 'orange', 'red']};
var vizBuilt = {min: 0, max: 100, palette: ['white', 'gray', 'black']};

// Add Layers
Map.addLayer(meanLST, vizLST, 'Land Surface Temp (C)');
Map.addLayer(meanNO2, vizNO2, 'Tropospheric NO2 Density');
Map.addLayer(ghsl, vizBuilt, 'GHSL Built-Up Surface');
🔑
Beginner note: This script defines a 25km buffer around central London and extracts environmental exposure predictors during summer 2023. Tropospheric NO2 column density is derived from Sentinel-5P to quantify air pollution levels. MODIS 8-day LST is converted from Kelvin scale factor values to degrees Celsius to assess urban heat exposure. Finally, GHSL built-up surface intensity is integrated to represent physical infrastructure density.

GeoAI Composite Vulnerability Index and Spatial Prediction
10 min

Add this block after the previous code. It continues the analysis with additional processing and visualization.

GeoAI Composite Vulnerability Index and Spatial Prediction
// Continuation: Min-Max Normalization and GeoAI Modeling

// Helper function for Min-Max Normalization
var normalize = function(img, geometry) {
  var minMax = img.reduceRegion({
    reducer: ee.Reducer.minMax(),
    geometry: geometry,
    scale: 1000,
    maxPixels: 1e9
  });
  var keys = minMax.keys();
  var minVal = ee.Number(minMax.get(keys.get(0)));
  var maxVal = ee.Number(minMax.get(keys.get(1)));
  return img.subtract(minVal).divide(maxVal.subtract(minVal)).rename('normalized');
};

var lstNorm = normalize(meanLST, roi);
var no2Norm = normalize(meanNO2, roi);
var builtNorm = normalize(ghsl, roi);

// Compute Composite Health Vulnerability Index (UVI)
var uvi = lstNorm.multiply(0.4)
  .add(no2Norm.multiply(0.4))
  .add(builtNorm.multiply(0.2))
  .rename('UVI');

// Load Elevation Covariate
var elevation = ee.Image('USGS/SRTMGL1_003').select('elevation').clip(roi);

// Stack features for GeoAI Modeling
var featureStack = meanLST.rename('LST')
  .addBands(meanNO2.rename('NO2'))
  .addBands(ghsl.rename('Built'))
  .addBands(elevation);

// Generate Stratified Random Training Sample
var samples = featureStack.sample({
  region: roi,
  scale: 1000,
  numPixels: 1000,
  geometries: true
});

// Train Random Forest Regressor to predict Thermal Risk (LST)
var rfModel = ee.Classifier.smileRandomForest(50).setOutputMode('REGRESSION');
var trainedModel = rfModel.train({
  features: samples,
  classProperty: 'LST',
  inputProperties: ['NO2', 'Built', 'elevation']
});

var predictedLST = featureStack.classify(trainedModel).rename('Predicted_LST');

// Visualization
var vizUVI = {min: 0, max: 0.8, palette: ['navy', 'cyan', 'lime', 'yellow', 'red']};
Map.addLayer(uvi, vizUVI, 'Urban Health Vulnerability Index (UVI)');
Map.addLayer(predictedLST, vizLST, 'GeoAI Predicted LST');
🔑
Beginner note: This script normalizes atmospheric NO2, land surface temperature, and built-up fraction using min-max scaling to build an Urban Health Vulnerability Index (UVI). Next, SRTM terrain elevation is added as an environmental driver. A Random Forest regression classifier with 50 decision trees is trained on random spatial samples to predict LST based on environmental and structural covariates, demonstrating spatial risk extrapolation.
